# Arbitrations: land that changes hands by the decision of a great power, not by war (decisions of the great powers' tables, 1938-39).
# The conditions say who can take it; the AI weights (what the country has done, who its friends are) say who does.

vienna_award = {
	name = "The Vienna Award"
	desc = "Hungary asks for the Magyar lands of southern Slovakia and Ruthenia. As the arbiter of Prague's fate, we can give them - and bind Budapest to us."
	potential = {
		NOT = { has_country_flag = vienna_award_done }
		exists = HUN
		ruling_ideology = fascist
		OR = { has_completed_focus = ger_demand_sudeten has_completed_focus = ger_czech_rump has_completed_focus = ger_anschluss NOT = { exists = AUS } is_subject_of = GER NOT = { exists = CZE } }
	}
	allow = { NOT = { is_at_war_with = HUN } NOT = { is_overlord_of = HUN } }
	effect = {
		transfer_state = { state = nitra_state target = HUN }
		transfer_state = { state = banska_bystrica_state target = HUN }
		transfer_state = { state = presov_state target = HUN }
		transfer_state = { state = carpathian_ruthenia_state target = HUN }
		add_opinion = { target = HUN value = 40 name = "Gave us back our land" }
		HUN = { add_opinion = { target = ROOT value = 40 name = "The Vienna Award" } }
		set_country_flag = vienna_award_done
	}
	ai_will_do = {
		base = 4
		modifier = { add = 3 any_neighbor_country = { tag = HUN } }
		modifier = { add = 3 is_allied_with = HUN }
		modifier = { factor = 0 NOT = { exists = HUN } }
	}
}
