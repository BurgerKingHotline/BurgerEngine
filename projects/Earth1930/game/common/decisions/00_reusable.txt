# Decisions (EU4 style). Reusable ones: the conditions pick who can take them and the AI weights (a country's personality, its ideology,
# its situation) pick who does - nothing here runs on a date. ai_will_do: the AI takes a decision with chance weight x 10% a month.

remilitarize_zone = {
	name = "Remilitarize the Zone"
	desc = "Our treaty obligations keep armed forces out of part of our own land. Marching in again will be a challenge to those who imposed it - a country that feels strong enough may dare."
	potential = { has_demilitarized_states = yes NOT = { is_at_war = yes } }
	allow = { num_of_divisions >= 3 }
	effect = { remilitarize_states = yes add_stability = 2 }
	repeatable = yes
	cooldown_days = 365
	ai_will_do = {
		base = 0
		modifier = { add = 3 remilitarization_odds >= 0.8 ai_aggression >= 1.0 }
		modifier = { add = 4 remilitarization_odds >= 1.2 ai_aggression >= 0.8 }
		modifier = { add = 3 remilitarization_odds >= 2 }
		modifier = { add = 2 ruling_ideology = fascist remilitarization_odds >= 0.7 }
		modifier = { factor = 0 remilitarization_odds < 0.55 }
	}
}

rearmament_programme = {
	name = "Rearmament Programme"
	desc = "Put the economy and the manpower of the nation behind the army."
	potential = { NOT = { has_modifier = rearmament } NOT = { is_at_war = yes } }
	allow = { political_power >= 40 }
	effect = { add_modifier = { name = rearmament years = 6 } add_political_power = -40 set_law = extensive_conscription }
	ai_will_do = {
		base = 0
		modifier = { add = 3 ai_aggression >= 1.2 }
		modifier = { add = 2 ruling_ideology = fascist }
		modifier = { add = 1.5 ruling_ideology = communist }
		modifier = { add = 2 any_neighbor_country = { stronger_than = { target = ROOT ratio = 1.4 } ideological_enemy = ROOT } }
		modifier = { add = 1.5 war_exhaustion < 1 ai_expansion >= 1.3 }
		modifier = { factor = 0.25 ai_isolation >= 1.5 }
		modifier = { factor = 0.5 stability < 40 }
	}
}

industrial_drive = {
	name = "Industrial Drive"
	desc = "State support for factories, railways and mines: new industry in the richest states."
	potential = { NOT = { has_modifier = industrial_drive } }
	allow = { political_power >= 30 stability >= 35 }
	effect = {
		add_modifier = { name = industrial_drive years = 4 }
		add_political_power = -30
		random_owned_state = { limit = { is_coastal = no } add_building = civilian_factory add_building = military_factory }
		random_owned_state = { add_building = civilian_factory }
	}
	repeatable = yes
	cooldown_days = 730
	ai_will_do = {
		base = 1
		modifier = { add = 2 commodity_shortage > 0.3 }
		modifier = { add = 1 num_of_divisions < 6 ai_expansion >= 1 }
		modifier = { factor = 0.2 stability < 40 }
	}
}
