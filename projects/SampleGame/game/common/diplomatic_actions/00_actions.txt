# Diplomatic actions (Imperator: Rome style) - shown in the country panel.
# Scopes:  ROOT / scope:actor = the country doing the action,  scope:recipient = the target country.
#   potential        shown only if true
#   allow            clickable only if true (each condition is listed in the tooltip)
#   cost             gold paid when used           cooldown_days   per target
#   requires_acceptance = yes  → the AI decides with ai_acceptance (> 0 = yes), the player gets a popup
#   ai_acceptance    base + modifiers; opinion_factor = recipient's opinion of us × factor;
#                    strength_factor = factor × (our military / their military - 1)
#   on_accept / on_decline   effects        ai_will_do   how much the AI wants to do it (0 = never)

improve_relations = {
	name = "Improve Relations"
	desc = "Send diplomats to improve their opinion of us (+25, decays over time)."
	cost = 10
	cooldown_days = 180
	potential = { NOT = { is_at_war_with = scope:recipient } }
	allow = { OR = { NOT = { has_feature = economy } gold >= 10 } }
	on_accept = {
		reverse_add_opinion = { target = scope:recipient value = 25 name = "Improved relations" }
	}
	ai_will_do = {
		base = 0
		modifier = { add = 2 scope:recipient = { opinion = { target = ROOT value < 0 } } }
		modifier = { add = 1 borders = scope:recipient }
	}
}

send_gift = {
	name = "Send Gift"
	desc = "Send 50 gold as a gift (+40 opinion)."
	cost = 50
	cooldown_days = 365
	potential = {
		has_feature = economy
		NOT = { is_at_war_with = scope:recipient }
	}
	allow = { gold >= 50 }
	on_accept = {
		scope:recipient = { add_gold = 50 }
		reverse_add_opinion = { target = scope:recipient value = 40 name = "Gift" }
	}
}

		reverse_add_opinion = { target = scope:recipient value = -40 name = "Insulted" }

offer_alliance = {
	name = "Offer Alliance"
	desc = "Allies are called to arms when either of us is attacked."
	requires_acceptance = yes
	potential = {
		NOT = { is_allied_with = scope:recipient }
		NOT = { is_at_war_with = scope:recipient }
	}
	allow = {
		free_relations >= 1
		scope:recipient = { free_relations >= 1 }
	}
	ai_acceptance = {
		base = -25
		opinion_factor = 0.5
		modifier = { add = 15 borders = scope:recipient }
		modifier = { add = -1000 scope:recipient = { is_at_war_with = ROOT } }
	}
	on_accept = {
		create_alliance = scope:recipient
		reverse_add_opinion = { target = scope:recipient value = 20 name = "Alliance" }
	}
	on_decline = {
		reverse_add_opinion = { target = scope:recipient value = -5 name = "Rejected alliance" }
	}
	ai_will_do = {
		base = 0
		modifier = { add = 1 scope:recipient = { opinion = { target = ROOT value > 25 } } }
	}
}

break_alliance = {
	name = "Break Alliance"
	potential = { is_allied_with = scope:recipient }
	ai_will_do = {
		base = 0
		modifier = { add = 2 opinion = { target = scope:recipient value < -20 } }
		# ambition: a strong ally next door that we have claims on is worth more as a conquest
		modifier = {
			add = 1
			has_claim_on = scope:recipient
			borders = scope:recipient
			stronger_than = { target = scope:recipient ratio = 1.5 }
		}
	}
	on_accept = {
		break_alliance = scope:recipient
		reverse_add_opinion = { target = scope:recipient value = -50 name = "Broke alliance" }
	}
}

fabricate_claim = {
	name = "Fabricate Claim"
	desc = "Forge documents proving one of their states is rightfully ours. Claims give a cheaper casus belli."
	cost = 30
	cooldown_days = 365
	potential = {
		borders = scope:recipient
		NOT = { is_allied_with = scope:recipient }
	}
	allow = { OR = { NOT = { has_feature = economy } gold >= 30 } }
	on_accept = {
		fabricate_claim = { target = scope:recipient }
		reverse_add_opinion = { target = scope:recipient value = -25 name = "Fabricated claims" }
	}
	ai_will_do = {
		base = 0
		modifier = { add = 1 military_strength > scope:recipient.military_strength }
		modifier = { factor = 0 is_at_war = yes }
	}
}

declare_war = {
	name = "Declare War"
	desc = "The best available casus belli is used (claims are cheaper in peace deals)."
	potential = {
		NOT = { is_at_war_with = scope:recipient }
		NOT = { is_allied_with = scope:recipient }
	}
	allow = {
		can_declare_war_on = scope:recipient
		num_of_divisions >= 1
	}
	on_accept = {
		declare_war = scope:recipient
	}
	ai_will_do = {
		base = 0
		modifier = {
			add = 4
			has_claim_on = scope:recipient
			borders = scope:recipient
			military_strength > scope:recipient.military_strength
		}
		# conquest: a much stronger neighbour attacks even without claims (less often)
		modifier = {
			add = 1
			borders = scope:recipient
			stronger_than = { target = scope:recipient ratio = 1.6 }
			NOT = { is_overlord_of = scope:recipient }
		}
		modifier = { factor = 0 is_at_war = yes }
	}
}

}
